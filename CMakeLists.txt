cmake_minimum_required(VERSION 3.20)
project(saw_dsu VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(saw_dsu INTERFACE)
target_include_directories(saw_dsu INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(saw_dsu INTERFACE Threads::Threads)
target_compile_options(saw_dsu INTERFACE -Wall -Wextra -Wpedantic)
add_executable(saw_bench src/benchmark.cpp)
target_link_libraries(saw_bench PRIVATE saw_dsu)
add_executable(saw_tests tests/unit.cpp)
target_link_libraries(saw_tests PRIVATE saw_dsu)
enable_testing()
add_test(NAME saw_unit COMMAND saw_tests)
